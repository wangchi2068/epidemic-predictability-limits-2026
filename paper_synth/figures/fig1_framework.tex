% ---------------- Tier 1: 双源随机性输入 ----------------
\node[card=themeTeal, fill=bgTeal, text width=6.2cm] (demo) {
    \textbf{\textcolor{themeTeal}{\large 动力学内在随机性}}\\[2pt]
    \textbf{\textcolor{themeTeal}{\footnotesize （传播过程噪声）}}\\[5pt]
    \footnotesize 负二项分支过程 $Z_{t+1} \mid Z_t \sim \mathrm{NB}(R Z_t,\, k)$\\[5pt]
    \footnotesize 微观相对方差饱和常数（\textbf{引理 2}）：\\[2pt]
    $\mathrm{CV}^2_\infty = \dfrac{1+R/k_{\mathrm{ind}}}{I_0(R-1)}$\\[5pt]
    \footnotesize 宏观周度更新几何发散项（\textbf{定理 2}）：\\[2pt]
    $\bigl[(1+1/k_{\mathrm{agg}})^h - 1\bigr]$
};

\node[card=themeDarkBlue, fill=bgBlue, text width=6.2cm, right=2.0cm of demo] (param) {
    \textbf{\textcolor{themeDarkBlue}{\large 参数推断不确定性}}\\[2pt]
    \textbf{\textcolor{themeDarkBlue}{\footnotesize （统计估计误差）}}\\[5pt]
    \footnotesize 监测窗口估计 $\log\hat{R} \sim \mathcal{N}(\log R,\, s^2_{\text{周}})$\\[5pt]
    \footnotesize 外推放大律（\textbf{引理 3}）：\\[2pt]
    $P_{\text{exact}}(h;s) = e^{2h^2s^2} - 2e^{\frac{1}{2}h^2s^2} + 1$\\[5pt]
    \footnotesize 代入式点预测系统性上偏：$\mathbb{E}[\hat{R}^h] = R^h e^{h^2s^2/2} > R^h$
};

% ---------------- Tier 2: 条件风险分解 ----------------
\node[card=themeSlate, fill=white, text width=12.8cm] (decomp) at ($(demo.south east)!0.5!(param.south west) + (0,-3.3)$) {
    \textbf{\textcolor{themeDarkBlue}{\large 条件风险分解（\textbf{定理 1} 锐条件风险恒等式；假定 1 条件正交）}}\\[7pt]
    $\mathrm{RelMSE}(h) \;=\; \underbrace{\mathrm{CV}^2_{\text{macro}}(h)}_{\text{内在机制贡献}} \;+\; \underbrace{P_{\text{exact}}(h;s)}_{\text{外在推断贡献}}$
    \quad \footnotesize （关于 $h$ 严格单调递增：定理 2、引理 3）\\[7pt]
    \footnotesize 目标相对误差预算阈值 $\tau$（公共卫生决策容忍度，如 $\tau=0.5$ 对应 50\% 相对误差上限）
};

% ---------------- Tier 3: 视界输出与决策映射 ----------------
\node[card=themeOrange, fill=bgOrange, text width=9.4cm] (horizon) at ($(decomp.south) + (0,-3.7)$) {
    \textbf{\textcolor{themeOrange}{\large 条件机制预测视界（\textbf{定理 4}）}}\\[6pt]
    \footnotesize 阈值根定义：\\[2pt]
    $h^{*}_{\text{cont}} = \sup\{\, h \ge 0 \mid \mathrm{RelMSE}(h) \le \tau^2 \,\}$\\[6pt]
    \footnotesize 线性化闭式速算：\\[2pt]
    $h^{*} \approx \dfrac{\sqrt{b^2 + 4s^2_{\text{周}}\tau^2} - b}{2s^2_{\text{周}}},\;\; b \equiv 1/k_{\mathrm{agg}}$\\[3pt]
    \footnotesize （较数值精确根高估 8\%--29\%，中位约 10\%；精确值以数值求根为准）
};

\node[card=black!45, fill=white, text width=5.6cm] (fisher) at ($(horizon.south) + (-3.15,-1.9)$) {
    \footnotesize\textbf{辨识样本量门槛（\textbf{定理 3}）}\\[3pt]
    \footnotesize Fisher 信息：$\mathcal{I}_N = \dfrac{Nk}{R(R+k)}$\\[3pt]
    \footnotesize 相对标准误达 $\varepsilon$ 需 $N \gtrsim O(\varepsilon^{-2})$
};

\node[card=black!45, fill=white, text width=5.6cm] (budget) at ($(horizon.south) + (3.15,-1.9)$) {
    \footnotesize\textbf{实测误差代数记账（第 4.3 节）}\\[3pt]
    \footnotesize $\mathrm{RelMSE}_{\text{obs}} = \mathrm{CV}^2_{\text{macro}} + P_{\text{exact}} + \mathcal{E}_{\text{res}}$\\[3pt]
    \footnotesize 21 配置差额占比中位数 55.0\%
};
